local function getHWID()
    if getexecutorname and syn and syn.request then
        return tostring(syn.get_hwid and syn.get_hwid() or "UNKNOWN")
    elseif getexecutorname then
        local name = string.lower(tostring(getexecutorname()))
        if name:find("delta") then
            return tostring(getcustomasset and getcustomasset("") or "UNKNOWN")
        end
    end
    if identifyexecutor then
        local n = identifyexecutor()
        return tostring(n or "UNKNOWN")
    end
    return "UNKNOWN"
end
